package com.example.jira.plugin.rest;

import com.atlassian.plugins.rest.common.security.AnonymousAllowed;
import org.apache.http.Header;
import org.apache.http.client.config.CookieSpecs;
import org.apache.http.client.config.RequestConfig;
import org.apache.http.client.methods.CloseableHttpResponse;
import org.apache.http.client.methods.HttpGet;
import org.apache.http.impl.client.CloseableHttpClient;
import org.apache.http.impl.client.HttpClients;
import org.apache.http.impl.conn.PoolingHttpClientConnectionManager;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import javax.servlet.http.HttpServletRequest;
import javax.ws.rs.GET;
import javax.ws.rs.Path;
import javax.ws.rs.core.Context;
import javax.ws.rs.core.Response;
import javax.ws.rs.core.StreamingOutput;
import java.io.IOException;
import java.io.InputStream;
import java.util.Enumeration;
import java.util.Set;

@Path("/proxy")
public class ProxyResource {

    private static final Logger log = LoggerFactory.getLogger(ProxyResource.class);
    
    private static final String TARGET_URL = "http://localhost:8080/jira/rest/api/2/target";
    private static final String ADMIN_TOKEN = "Bearer YOUR_ADMIN_TOKEN";
    
    private static final Set<String> BLOCKED_HEADERS = Set.of(
        "cookie", "set-cookie", "set-cookie2", "authorization", 
        "connection", "keep-alive", "proxy-authenticate", "proxy-authorization", 
        "te", "trailers", "transfer-encoding", "upgrade", 
        "content-length", "content-encoding"
    );

    private static final CloseableHttpClient httpClient = createHttpClient();

    @GET
    @Path("/stream")
    @AnonymousAllowed
    public Response proxy(@Context HttpServletRequest clientRequest) {
        HttpGet upstreamRequest = createUpstreamRequest(clientRequest);

        try {
            CloseableHttpResponse upstreamResponse = httpClient.execute(upstreamRequest);
            return createDownstreamResponse(upstreamResponse);
        } catch (Exception e) {
            log.error("Proxy execution failed", e);
            return Response.serverError().build();
        }
    }

    private HttpGet createUpstreamRequest(HttpServletRequest clientRequest) {
        HttpGet request = new HttpGet(TARGET_URL);
        
        request.setHeader("Authorization", ADMIN_TOKEN);
        request.setHeader("Accept-Encoding", "identity");

        Enumeration<String> headerNames = clientRequest.getHeaderNames();
        while (headerNames.hasMoreElements()) {
            String name = headerNames.nextElement();
            if (isAllowedHeader(name)) {
                request.addHeader(name, clientRequest.getHeader(name));
            }
        }
        return request;
    }

    private Response createDownstreamResponse(CloseableHttpResponse upstreamResponse) throws IOException {
        int statusCode = upstreamResponse.getStatusLine().getStatusCode();
        Response.ResponseBuilder response = Response.status(statusCode);

        for (Header header : upstreamResponse.getAllHeaders()) {
            if (isAllowedHeader(header.getName())) {
                response.header(header.getName(), header.getValue());
            }
        }

        if (upstreamResponse.getEntity() != null) {
            response.entity(createStreamingOutput(upstreamResponse.getEntity().getContent()));
        }

        return response.build();
    }

    private StreamingOutput createStreamingOutput(InputStream inputStream) {
        return outputStream -> {
            try (InputStream in = inputStream) {
                in.transferTo(outputStream);
            }
        };
    }

    private boolean isAllowedHeader(String name) {
        return name != null && !name.isBlank() && !BLOCKED_HEADERS.contains(name.trim().toLowerCase());
    }

    private static CloseableHttpClient createHttpClient() {
        PoolingHttpClientConnectionManager cm = new PoolingHttpClientConnectionManager();
        cm.setMaxTotal(200);
        cm.setDefaultMaxPerRoute(20);

        RequestConfig config = RequestConfig.custom()
                .setConnectTimeout(5000)
                .setSocketTimeout(10000)
                .setCookieSpec(CookieSpecs.IGNORE_COOKIES)
                .build();

        return HttpClients.custom()
                .setConnectionManager(cm)
                .setDefaultRequestConfig(config)
                .disableContentCompression()
                .build();
    }
}

지라 알림 메일 처리라는 비즈니스 로직의 특성을 살려 **가독성을 최우선**으로 리팩토링한 코드입니다.
기존 코드의 자잘한 중복(바이트 계산 등)을 없애고, 자바의 가독성을 높여주는 패턴을 적용하여 **"코드가 마치 한 편의 설명서처럼 읽히도록"** 다듬었습니다.
### 🛠️ 가독성 최우선 리팩토링 코드
```java
import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;

import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class JiraEmailSanitizer {

    // 용량 제한 정책 설정 (1.8MB 마진 확보)
    private static final long MAX_ALLOWED_BYTES = 1887436; 
    private static final int LARGE_IMAGE_STRING_LIMIT = 133000; 

    public static void main(String[] args) {
        // 테스트용 샘플 HTML 데이터 (생략)
        String jiraHtml = "<html><body>...</div></body></html>";
        String processedHtml = sanitize(jiraHtml);
    }

    /**
     * [메인 엔트리] 지라 메일의 용량을 검사하고 필요 시 단계별로 다이어트를 수행합니다.
     */
    public static String sanitize(String originalHtml) {
        Document document = Jsoup.parse(originalHtml);
        long initialBytes = calculateByteSize(document);
        
        // 1. 용량이 안정권이라면 아무 작업 없이 원본 반환 (Early Return)
        if (initialBytes <= MAX_ALLOWED_BYTES) {
            return originalHtml;
        }

        // 2. 초과 시 단계별 순차 다이어트 실행 (용량 충족 시 중간에 알아서 멈춤)
        reduceLargeImages(document);
        removeOlderComments(document);
        removeActivityLogs(document);
        enforceHardLimitAsLastResort(document);

        // 3. 내용이 하나라도 유실되었다면 안내 배너 주입
        if (calculateByteSize(document) < initialBytes) {
            injectWarningBanner(document);
        }

        return document.outerHtml();
    }

    /**
     * [1단계] 본문 내 대용량 인라인 이미지(Base64)를 가벼운 안내 문구로 치환합니다.
     */
    private static void reduceLargeImages(Document doc) {
        Elements images = doc.select("img[src^=data:]");
        
        for (Element img : images) {
            if (isSizeSafe(doc)) return; // 용량이 이미 안전하면 조기 종료

            String src = img.attr("src");
            if (src.length() > LARGE_IMAGE_STRING_LIMIT) {
                img.replaceWith(createImageOmitNotice(src.length()));
            }
        }
    }

    /**
     * [2단계] 과거 댓글(table.action-container)을 하단(오래된 순)부터 차례대로 삭제합니다.
     */
    private static void removeOlderComments(Document doc) {
        if (isSizeSafe(doc)) return;

        List<Element> comments = new ArrayList<>(doc.select("table.action-container"));
        Collections.reverse(comments); // 하단부터 지우기 위해 리스트 반전

        for (Element comment : comments) {
            if (isSizeSafe(doc)) return;
            comment.remove(); 
        }
    }

    /**
     * [3단계] 활동 로그(table.activity-stream) 영역을 삭제합니다.
     */
    private static void removeActivityLogs(Document doc) {
        if (isSizeSafe(doc)) return;

        Elements activityStreams = doc.select("table.activity-stream");
        for (Element stream : activityStreams) {
            if (isSizeSafe(doc)) return;
            stream.remove();
        }
    }

    /**
     * [4단계] 최후의 보루: 모든 수단을 썼음에도 초과일 때, 본문 하단 요소부터 강제 무차별 삭제합니다.
     */
    private static void enforceHardLimitAsLastResort(Document doc) {
        if (isSizeSafe(doc)) return;

        List<Element> bodyChildren = new ArrayList<>(doc.body().children());
        Collections.reverse(bodyChildren);
        
        for (Element child : bodyChildren) {
            if (isSizeSafe(doc)) return;
            child.remove();
        }
    }

    // =================================================================
    //  가독성 및 캡슐화를 위한 가공용 서브 헬퍼(Helper) 메서드들
    // =================================================================

    /**
     * 현재 문서가 안전 용량 기준(2MB 이하)을 만족하는지 여부를 반환합니다.
     */
    private static boolean isSizeSafe(Document doc) {
        return calculateByteSize(doc) <= MAX_ALLOWED_BYTES;
    }

    /**
     * Document의 현재 실시간 UTF-8 바이트 크기를 구합니다.
     */
    private static long calculateByteSize(Document doc) {
        return doc.outerHtml().getBytes(StandardCharsets.UTF_8).length;
    }

    /**
     * 잘려나간 대용량 이미지 자리에 들어갈 아웃룩 호환 HTML 안내 박스를 생성합니다.
     */
    private static Element createImageOmitNotice(int stringLength) {
        int estimatedKB = (int) ((stringLength * 0.75) / 1024);
        String html = String.format(
            "<div style='display:inline-block; padding:12px; background-color:#f4f5f7; " +
            "border:1px dashed #dfe1e6; color:#5e6c84; font-size:12px; margin:5px 0; border-radius:3px; font-family:sans-serif;'>" +
            "🖼️ [본문 삽입 대용량 이미지 생략 (%dKB) - Jira 시스템에서 확인하세요]" +
            "</div>", estimatedKB
        );
        return Jsoup.parseBodyFragment(html).body().child(0);
    }

    /**
     * 메일 최상단에 유실 안내 컴포넌트를 주입합니다.
     */
    private static void injectWarningBanner(Document doc) {
        String bannerHtml = 
            "<div style='color:#a51d24; font-weight:bold; padding:12px; " +
            "border:1px solid #fbe3e4; background-color:#fdf2f2; margin-bottom:15px; " +
            "font-size:13px; border-radius:4px; font-family:sans-serif;'>" +
            "⚠️ 본 알림 메일은 시스템 발송 제한 용량(2MB)을 초과하여, 최신 정보 유지를 위해 하단의 과거 댓글 또는 대용량 이미지가 일부 생략되었습니다. " +
            "전체 내용은 Jira 시스템에 접속하여 확인해 주세요.</div>";
        doc.body().prepend(bannerHtml);
    }
}

```
### 🌟 무엇이 개선되었나요? (가독성 포인트)
 1. **지저분한 if 중첩 제거 (Early Return)**
   * 메인 메소드(sanitize) 시작 시 용량이 이미 정상이라면 아래의 복잡한 4단계 함수를 읽을 필요도 없이 바로 원본을 리턴(return originalHtml;)하게 하여 메인 흐름을 극도로 단순화했습니다.
 2. **비즈니스 언어로 된 서술형 메서드 (isSizeSafe)**
   * 기존의 if (getByteLength(doc) <= MAX_BYTES) break; 같은 수식 형태의 복잡한 비교문을 if (isSizeSafe(doc)) return; 이라는 **직관적인 문장형태**로 교체했습니다. 코드를 읽는 속도가 훨씬 빨라집니다.
 3. **불필요한 로그(System.out.println) 제거**
   * 실서비스 운영 코드에서 가독성을 해치는 콘솔 디버그용 출력 코드를 걷어내고 순수 로직만 남겼습니다.
 4. **HTML 템플릿 코드 분리**
   * 메인 로직 중간에 길게 늘어져 가독성을 해치던 replacementHtml 생성 코드를 createImageOmitNotice() 함수로 격리하여 핵심 다이어트 로직의 순도를 높였습니다.

**Java 11+ 표준 API (java.net.http.HttpClient)**를 사용하여 외부 라이브러리 의존성을 제거하고, 가독성과 안정성을 극대화한 최종 리팩토링 버전입니다.
Jira 9.x 환경에서 가장 권장되는 Pure Java 구현입니다.
package com.example.jira.plugin.rest;

import com.atlassian.plugins.rest.common.security.AnonymousAllowed;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import javax.servlet.http.HttpServletRequest;
import javax.ws.rs.GET;
import javax.ws.rs.Path;
import javax.ws.rs.core.Context;
import javax.ws.rs.core.Response;
import javax.ws.rs.core.StreamingOutput;
import java.io.IOException;
import java.io.InputStream;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.time.Duration;
import java.util.Enumeration;
import java.util.Set;

@Path("/proxy")
public class NativeProxyResource {

    private static final Logger log = LoggerFactory.getLogger(NativeProxyResource.class);

    private static final String TARGET_URL = "http://localhost:8080/jira/rest/api/2/target";
    private static final String ADMIN_TOKEN = "Bearer YOUR_ADMIN_TOKEN";

    private static final Set<String> BLOCKED_HEADERS = Set.of(
        "cookie", "set-cookie", "set-cookie2", "authorization",
        "connection", "keep-alive", "proxy-authenticate", "proxy-authorization",
        "te", "trailers", "transfer-encoding", "upgrade",
        "content-length", "content-encoding"
    );

    private static final HttpClient httpClient = createHttpClient();

    @GET
    @Path("/stream")
    @AnonymousAllowed
    public Response proxy(@Context HttpServletRequest clientRequest) {
        HttpRequest upstreamRequest = createUpstreamRequest(clientRequest);

        try {
            HttpResponse<InputStream> upstreamResponse = httpClient.send(upstreamRequest, HttpResponse.BodyHandlers.ofInputStream());
            return createDownstreamResponse(upstreamResponse);
        } catch (Exception e) {
            log.error("Proxy execution failed", e);
            return Response.serverError().build();
        }
    }

    private HttpRequest createUpstreamRequest(HttpServletRequest clientRequest) {
        HttpRequest.Builder builder = HttpRequest.newBuilder()
                .uri(URI.create(TARGET_URL))
                .header("Authorization", ADMIN_TOKEN)
                .header("Accept-Encoding", "identity")
                .GET();

        Enumeration<String> headerNames = clientRequest.getHeaderNames();
        while (headerNames.hasMoreElements()) {
            String name = headerNames.nextElement();
            if (isAllowedHeader(name)) {
                builder.header(name, clientRequest.getHeader(name));
            }
        }
        return builder.build();
    }

    private Response createDownstreamResponse(HttpResponse<InputStream> upstreamResponse) {
        Response.ResponseBuilder response = Response.status(upstreamResponse.statusCode());

        upstreamResponse.headers().map().forEach((key, values) -> {
            if (isAllowedHeader(key)) {
                values.forEach(value -> response.header(key, value));
            }
        });

        response.entity(createStreamingOutput(upstreamResponse.body()));
        return response.build();
    }

    private StreamingOutput createStreamingOutput(InputStream inputStream) {
        return outputStream -> {
            try (InputStream in = inputStream) {
                in.transferTo(outputStream);
            }
        };
    }

    private boolean isAllowedHeader(String name) {
        return name != null && !name.isBlank() && !BLOCKED_HEADERS.contains(name.trim().toLowerCase());
    }

    private static HttpClient createHttpClient() {
        return HttpClient.newBuilder()
                .connectTimeout(Duration.ofSeconds(5))
                .followRedirects(HttpClient.Redirect.NORMAL)
                .build();
    }
}
